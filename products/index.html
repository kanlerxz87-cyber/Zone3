for a in B04G1MEwY B02NbDxp3 B00hJ57DV; do echo "$a chars=${#a}" curl -sSI -A "Agent/Zone3ProductResearch" "https://www.amazon.com/dp/$a" \ | head -1 done # control (should serve a real page): curl -sSI -A "Agent/Zone3ProductResearch" https://www.amazon.com/dp/B0CDY277D3 | head -1
